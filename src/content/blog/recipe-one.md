servings: 8

prepTime: "15 mins"
cookTime: "50 mins"
ingredients:
  - "3 ripe bananas, mashed"
  - "1/3 cup melted butter"
  - "3/4 cup granulated sugar"
  - "1 large egg, beaten"
  - "1 tsp vanilla extract"
  - "1 tsp baking soda"
  - "Pinch of salt"
  - "1 1/2 cups all-purpose flour"
instructions:
  - "Preheat oven to 350°F (175°C) and grease a 4x8-inch loaf pan."
  - "In a mixing bowl, mash the ripe bananas until smooth."
  - "Stir the melted butter into the mashed bananas."
  - "Mix in the sugar, egg, and vanilla extract."
  - "Sprinkle in baking soda and salt, then fold in flour until combined."
  - "Pour batter into prepared loaf pan and bake for 50 to 60 minutes."
